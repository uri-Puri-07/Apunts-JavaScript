3. MÀXIM / MÍNIM

Math.max, Math.min i funcions pròpies.

Amb array

Math.max(...nums);
Math.min(...nums);

Funció màxim

function maxim(a, b) {
  if (a > b) {
    return a;
  } else {
    return b;
  }
}

Funció mínim

function minim(a, b) {
  if (a < b) {
    return a;
  } else {
    return b;
  }
}

4. CONDICIONS I SWITCH

}

Boolean

switch

switch (operacio) {
  case "+":

  default:
    console.log("Error");
}

11. FRUITES

createElement, classList, append, addEventListener, funció random i posició aleatòria.

Funció random

function generarAleatori(min, max) {
  return Math.floor(Math.random() * (max - min + 1)) + min;
}

Crear fruita

let f = document.createElement("div");
f.classList.add("fruita");
f.textContent = "🍎";

Posició aleatòria

let posX = generarAleatori(0, zonaJoc.clientWidth - 60);
let posY = generarAleatori(0, zonaJoc.clientHeight - 60);

f.style.left = posX + "px";
f.style.top = posY + "px";

Click per sumar punts

f.addEventListener("click", function() {
  f.remove();
  punts += 10;
  divPunts.textContent = punts;

13. LLISTA DINÀMICA, FILTRE I DRAG & DROP

Afegir elements a una llista, filtrar contingut i moure elements amb drag and drop.

Llista dinàmica

let input = document.getElementById("item-input");
let addButton = document.getElementById("add-button");
let list = document.getElementById("dynamic-list");

addButton.addEventListener("click", function () {
  if (input.value.trim() !== "") {
    let newItem = document.createElement("li");
    newItem.innerText = input.value;
    list.appendChild(newItem);
    input.value = "";
  }
});

Filtre

filterInput.addEventListener("keyup", function () {
  const term = filterInput.value.toLowerCase();
  const items = itemsList.getElementsByTagName("li");

  Array.from(items).forEach(function (item) {
    item.style.display =
      item.textContent.toLowerCase().includes(term)
        ? "block"
        : "none";
  });
});

Drag & Drop

draggable.addEventListener("dragstart", function () {
  setTimeout(() => {
    this.style.display = "none";
  }, 0);
});

dropzone.addEventListener("dragover", function (e) {
  e.preventDefault();
});

dropzone.addEventListener("drop", function () {
  draggable.style.display = "block";
  this.append(draggable);
});
